-- 1. 创建数据库及表结构
DROP DATABASE IF EXISTS school;
CREATE DATABASE school;
USE school;

-- 学院表
CREATE TABLE Department (
    Dno INT PRIMARY KEY AUTO_INCREMENT,
    Dname VARCHAR(50) NOT NULL UNIQUE,
    Dean VARCHAR(20)
);

-- 学生表
CREATE TABLE Student (
    Sno CHAR(9) PRIMARY KEY,
    Sname VARCHAR(20) NOT NULL,
    Ssex CHAR(1) CHECK (Ssex IN ('男','女')),
    Sage TINYINT,
    Sphone CHAR(11),
    Email VARCHAR(50),
    Dno INT,
    FOREIGN KEY (Dno) REFERENCES Department(Dno)
);

-- 课程表
CREATE TABLE Course (
    Cno CHAR(7) PRIMARY KEY,
    Cname VARCHAR(50) NOT NULL,
    Credit TINYINT,
    Dno INT,
    FOREIGN KEY (Dno) REFERENCES Department(Dno)
);

-- 教师表
CREATE TABLE Teacher (
    Tno CHAR(6) PRIMARY KEY,
    Tname VARCHAR(20) NOT NULL,
    Title VARCHAR(20),
    Dno INT,
    FOREIGN KEY (Dno) REFERENCES Department(Dno)
);

-- 选课表
CREATE TABLE SC (
    Sno CHAR(9),
    Cno CHAR(7),
    Tno CHAR(6),
    Usual_grade DECIMAL(5,2) CHECK (Usual_grade BETWEEN 0 AND 100),
    Final_grade DECIMAL(5,2) CHECK (Final_grade BETWEEN 0 AND 100),
    PRIMARY KEY (Sno, Cno),
    FOREIGN KEY (Sno) REFERENCES Student(Sno),
    FOREIGN KEY (Cno) REFERENCES Course(Cno),
    FOREIGN KEY (Tno) REFERENCES Teacher(Tno)
);

-- 3. 执行操作命令
-- (1) 建立计算机学院不及格学生视图
CREATE OR REPLACE VIEW V_Computer_Fail AS
SELECT s.Sno, s.Sname, s.Ssex, s.Sphone, c.Cname, 
       ROUND(sc.Usual_grade*0.3 + sc.Final_grade*0.7, 1) AS Total
FROM Student s
JOIN SC sc ON s.Sno = sc.Sno
JOIN Course c ON sc.Cno = c.Cno
WHERE s.Dno = 1 
  AND sc.Usual_grade IS NOT NULL 
  AND sc.Final_grade IS NOT NULL
  AND (sc.Usual_grade*0.3 + sc.Final_grade*0.7) < 60;

-- 查看视图结果
SELECT '操作1结果：计算机学院不及格学生视图' AS Operation;
SELECT * FROM V_Computer_Fail;

-- (2) 为每个学生插入未选课程
INSERT INTO SC (Sno, Cno, Tno)
SELECT s.Sno, c.Cno, 
  (SELECT Tno FROM Teacher WHERE Dno = c.Dno ORDER BY RAND() LIMIT 1)
FROM Student s
CROSS JOIN Course c
WHERE NOT EXISTS (
  SELECT 1 FROM SC 
  WHERE SC.Sno = s.Sno AND SC.Cno = c.Cno
);

-- 验证插入结果
SELECT '操作2结果：新增选课记录数' AS Operation;
SELECT COUNT(*) AS 新增记录数 FROM SC WHERE Usual_grade IS NULL AND Final_grade IS NULL;

-- (3) 查询年龄>所有女生的男生
SELECT '操作3结果：年龄大于所有女生的男生' AS Operation;
SELECT Sname, Sage
FROM Student
WHERE Ssex = '男'
  AND Sage > ALL (SELECT Sage FROM Student WHERE Ssex = '女');

    ELSE Usual_grade * 1.04
  END

-- (5) 删除没有开课的学院
DELETE FROM Department
WHERE Dno NOT IN (SELECT DISTINCT Dno FROM Course WHERE Dno IS NOT NULL);

-- 验证删除结果
SELECT '操作5结果：剩余学院列表' AS Operation;
SELECT Dno, Dname FROM Department;

-- (6) 查询成绩等级分布
SELECT '操作6结果：成绩等级分布' AS Operation;
SELECT 
  CASE
    WHEN total >= 90 THEN '优'
    WHEN total >= 80 THEN '良'
    WHEN total >= 70 THEN '中'
    WHEN total >= 60 THEN '及格'
    ELSE '不及格'
  END AS 等级,
  COUNT(*) AS 人数
FROM (
  SELECT ROUND(Usual_grade*0.3 + Final_grade*0.7, 1) AS total
  FROM SC
  WHERE Usual_grade IS NOT NULL 
    AND Final_grade IS NOT NULL
) AS grades
GROUP BY 等级
ORDER BY FIELD(等级, '优','良','中','及格','不及格');

  (SELECT COUNT(*) FROM Teacher) AS 教师数,
  (SELECT COUNT(*) FROM SC) AS 选课记录数;

-- 1. 修改学生表添加状态字段
ALTER TABLE Student ADD COLUMN Status VARCHAR(10) DEFAULT '正常' CHECK (Status IN ('正常', '试读', '退学'));

-- 2. 在选课表中添加学期字段
ALTER TABLE SC ADD COLUMN Semester VARCHAR(15) NOT NULL DEFAULT '2023-2024-1';

-- 3. 更新现有选课记录的学期信息
UPDATE SC SET Semester = CASE 
    WHEN RAND() < 0.3 THEN '2022-2023-2'  -- 30%为上学期
    ELSE '2023-2024-1'                     -- 70%为最近学期
END;

-- 4. 创建视图计算学生学期学分情况
CREATE VIEW V_Student_Credits AS
SELECT 
    s.Sno,
    s.Sname,
    sc.Semester,
    SUM(c.Credit) AS Total_Credits,
    SUM(CASE WHEN (sc.Usual_grade * 0.3 + sc.Final_grade * 0.7) < 60 THEN c.Credit ELSE 0 END) AS Fail_Credits
FROM Student s
JOIN SC sc ON s.Sno = sc.Sno
JOIN Course c ON sc.Cno = c.Cno
WHERE sc.Usual_grade IS NOT NULL 
  AND sc.Final_grade IS NOT NULL
GROUP BY s.Sno, sc.Semester;

-- 5. 更新试读状态
UPDATE Student
SET Status = '试读'
WHERE Sno IN (
    SELECT Sno
    FROM V_Student_Credits
    WHERE Semester = '2023-2024-1'  -- 最近学期
      AND Fail_Credits >= Total_Credits * 0.5
);

-- 6. 查看试读学生名单
SELECT '试读学生名单' AS Result;
SELECT 
    s.Sno, 
    s.Sname, 
    d.Dname AS College,
    v.Total_Credits,
    v.Fail_Credits,
    ROUND(v.Fail_Credits * 100.0 / v.Total_Credits, 1) AS Fail_Percent
FROM Student s
JOIN V_Student_Credits v ON s.Sno = v.Sno
JOIN Department d ON s.Dno = d.Dno
WHERE s.Status = '试读'
  AND v.Semester = '2023-2024-1';

-- 7. 查看所有学生状态分布
SELECT '学生状态统计' AS Result;
SELECT 
    Status,
    COUNT(*) AS Student_Count
FROM Student
GROUP BY Status;
