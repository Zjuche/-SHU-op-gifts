-- 创建数据库
CREATE DATABASE LibraryDB;
USE LibraryDB;

-- 员工表
CREATE TABLE EMPLOYEE (
    EmployeeID VARCHAR(10) PRIMARY KEY,
    Name VARCHAR(50) NOT NULL,
    Phone VARCHAR(15),
    Email VARCHAR(50) NOT NULL
);

-- 书目表
CREATE TABLE CATALOG (
    ISBN VARCHAR(20) PRIMARY KEY,
    Title VARCHAR(100) NOT NULL,
    Author VARCHAR(50) NOT NULL,
    Publisher VARCHAR(50) NOT NULL,
    PubDate DATE NOT NULL,
    TotalCount INT NOT NULL
);

-- 图书表
CREATE TABLE BOOK (
    BookID INT AUTO_INCREMENT PRIMARY KEY,
    ISBN VARCHAR(20) NOT NULL,
    Location VARCHAR(50) NOT NULL,
    Status ENUM('未借出','已借出','已预约','不外借') DEFAULT '未借出',
    FOREIGN KEY (ISBN) REFERENCES CATALOG(ISBN)
);

-- 借阅记录表
CREATE TABLE BORROW_RECORD (
    BorrowID INT AUTO_INCREMENT PRIMARY KEY,
    EmployeeID VARCHAR(10) NOT NULL,
    BookID INT NOT NULL,
    BorrowDate DATE NOT NULL,
    DueDate DATE NOT NULL,
    ReturnDate DATE,
    Fine DECIMAL(10,2) DEFAULT 0.00,
    FOREIGN KEY (EmployeeID) REFERENCES EMPLOYEE(EmployeeID),
    FOREIGN KEY (BookID) REFERENCES BOOK(BookID)
);

-- 预约记录表
CREATE TABLE RESERVATION (
    ReserveID INT AUTO_INCREMENT PRIMARY KEY,
    EmployeeID VARCHAR(10) NOT NULL,
    ISBN VARCHAR(20) NOT NULL,
    ReserveDate DATE NOT NULL,
    ExpireDate DATE NOT NULL,
    BookID INT,
    FOREIGN KEY (EmployeeID) REFERENCES EMPLOYEE(EmployeeID),
    FOREIGN KEY (ISBN) REFERENCES CATALOG(ISBN),
    FOREIGN KEY (BookID) REFERENCES BOOK(BookID)
);

-- 系统配置表
CREATE TABLE SYSTEM_CONFIG (
    ConfigID INT PRIMARY KEY DEFAULT 1,
    FinePerDay DECIMAL(10,2) NOT NULL DEFAULT 1.00,
    MaxBorrowDays INT NOT NULL DEFAULT 60,
    MaxReserveDays INT NOT NULL DEFAULT 7
);

-- 4. 还书操作（含超期罚款）
START TRANSACTION;
UPDATE BORROW_RECORD 
SET ReturnDate = CURDATE(),
    Fine = (SELECT FinePerDay FROM SYSTEM_CONFIG) * 
           DATEDIFF(CURDATE(), '2023-12-01')
WHERE BookID = 4 AND ReturnDate IS NULL;

-- 检查预约并分配图书
UPDATE RESERVATION SET BookID = 4 
WHERE ISBN = (SELECT ISBN FROM BOOK WHERE BookID = 4)
  AND BookID IS NULL
  AND ExpireDate >= CURDATE()
ORDER BY ReserveDate
LIMIT 1;

UPDATE BOOK SET Status = 
  IF((SELECT COUNT(*) FROM RESERVATION WHERE BookID = 4) > 0, 
     '已预约', '未借出')
WHERE BookID = 4;
COMMIT;

-- 5. 通知处理
-- 到期未归还通知
SELECT e.Email, b.BookID, c.Title, br.DueDate
FROM BORROW_RECORD br
JOIN EMPLOYEE e ON br.EmployeeID = e.EmployeeID
JOIN BOOK b ON br.BookID = b.BookID
JOIN CATALOG c ON b.ISBN = c.ISBN
WHERE br.ReturnDate IS NULL AND br.DueDate < CURDATE();

-- 预约到书通知
SELECT r.EmployeeID, e.Email, c.Title, b.BookID
FROM RESERVATION r
JOIN EMPLOYEE e ON r.EmployeeID = e.EmployeeID
JOIN CATALOG c ON r.ISBN = c.ISBN
JOIN BOOK b ON r.BookID = b.BookID
WHERE r.BookID IS NOT NULL 
  AND r.ExpireDate >= CURDATE()
  AND b.Status = '已预约';
